\chapter{Annotated Reading List}\label{app:reading}

Grouped by phase. \textbf{[M]} must-read for this project, \textbf{[U]} useful
when you reach that point, \textbf{[D]} deep-dive if you want more. Full
details are in the bibliography.

\section*{Foundations (Part I)}
\begin{itemize}
  \item \textbf{[M]} Harris \& Harris, \emph{Digital Design and Computer
        Architecture}~\cite{harris2021ddca} --- the clearest introduction to
        digital logic and HDLs.
  \item \textbf{[M]} Parhi, \emph{VLSI Digital Signal Processing
        Systems}~\cite{parhi1999vlsidsp} --- pipelining, folding, retiming:
        the vocabulary of DSP architecture.
  \item \textbf{[U]} Kaeslin, \emph{Top-Down Digital VLSI
        Design}~\cite{ch12-13:kaeslin2014topdown} --- architecture to gates,
        with an engineer's view of trade-offs.
  \item \textbf{[U]} Horowitz, \enquote{Computing's energy
        problem}~\cite{ch12-13:horowitz2014energy} --- why memory access
        dominates energy.
  \item \textbf{[D]} Weste \& Harris~\cite{weste2010cmos}; Rabaey
        \etal~\cite{rabaey2003digital} --- CMOS circuits in depth.
\end{itemize}

\section*{The thesis algorithms}
\begin{itemize}
  \item \textbf{[M]} Cohen, IMCRA~\cite{cohen2003imcra}; Ephraim \&
        Malah~\cite{ephraim1984mmse} --- re-read with \enquote{what does this cost
        in hardware?} in mind.
  \item \textbf{[U]} Loizou, \emph{Speech Enhancement}~\cite{loizou2013se};
        Martin, minimum statistics~\cite{martin2001minstat}.
  \item \textbf{[U]} Stone \& Moore on tolerable hearing-aid
        delay~\cite{stone1999delay,stone2008delay} --- the evidence behind RQ2.
  \item \textbf{[D]} Kates, \emph{Digital Hearing Aids}~\cite{kates2008hearingaids}.
\end{itemize}

\section*{Phase 1 --- fixed point and co-design}
\begin{itemize}
  \item \textbf{[M]} Widrow \& Koll\'ar, \emph{Quantization
        Noise}~\cite{widrow2008quantization} --- when the uniform-noise model
        holds and when it does not.
  \item \textbf{[U]} Constantinides \etal, word-length
        optimisation~\cite{ch12-13:constantinides2003wordlength}.
  \item \textbf{[U]} Muller, \emph{Elementary Functions}~\cite{muller2016elementary}
        --- tables, polynomials and Newton--Raphson done properly.
  \item \textbf{[U]} Oppenheim \& Schafer~\cite{oppenheim2010dtsp} --- FFT
        algorithms, real-input FFTs.
\end{itemize}

\section*{Phases 2--3 --- RTL and verification}
\begin{itemize}
  \item \textbf{[M]} Sutherland \etal, \emph{SystemVerilog for
        Design}~\cite{sutherland2006svdesign}; Cummings on non-blocking
        assignments~\cite{cummings2000nonblocking}.
  \item \textbf{[M]} cocotb documentation~\cite{ch12-13:cocotbdocs} and the
        Verilator guide~\cite{ch12-13:verilatorguide}.
  \item \textbf{[U]} Garrido, survey of pipelined FFT
        architectures~\cite{ch12-13:garrido2022fft}; He \& Torkelson,
        R2\textsuperscript{2}SDF~\cite{he1996r22sdf}.
  \item \textbf{[U]} HDLBits~\cite{ch12-13:hdlbits} for Verilog practice;
        Project~F~\cite{ch12-13:projectf} for FPGA tutorials; the ZipCPU
        blog~\cite{ch12-13:zipcpu} for formal verification of real designs.
  \item \textbf{[U]} SymbiYosys documentation~\cite{ch12-13:sbydocs}.
  \item \textbf{[D]} Spear \& Tumbush, \emph{SystemVerilog for
        Verification}~\cite{ch12-13:spear2012svverif}.
\end{itemize}
